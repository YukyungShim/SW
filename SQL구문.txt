  ### 📋 1. Supabase SQL Editor에 실행할 최종 SQL
  복잡한 함수 없이 아래 4가지 구문만 SQL 에디터에 넣고 실행하시면 됩니다:
    -- 1. 네온 웜 서바이벌 랭킹 테이블 생성
    CREATE TABLE IF NOT EXISTS public.neon_worm_leaderboard (
        id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
        -- 파일럿 닉네임 (12자 제한)
        name VARCHAR(12) NOT NULL,
        -- 생존 시간 (밀리초 단위, 0 이상)
        time_ms BIGINT NOT NULL CHECK (time_ms >= 0),
        -- 먹은 사과 수
        apples INTEGER NOT NULL DEFAULT 0 CHECK (apples >= 0),
        -- 점수
        score INTEGER NOT NULL DEFAULT 0 CHECK (score >= 0),
        -- 도달 위험 레벨 (1 이상)
        hazard_level INTEGER NOT NULL DEFAULT 1 CHECK (hazard_level > 0),
        played_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
        
        -- 한 명의 파일럿은 단 하나의 '최고 기록'만 유지하도록 고유키 설정
        UNIQUE (name)
    );
    
    -- 2. 랭킹 조회를 위한 정렬 인덱스 생성
    -- 생존 시간(내림차순) -> 사과 수(내림차순) -> 달성일(오름차순)
    CREATE INDEX IF NOT EXISTS idx_neon_worm_rank 
    ON public.neon_worm_leaderboard (time_ms DESC, apples DESC, played_at ASC);
    
    -- 3. Row Level Security (RLS) 활성화
    ALTER TABLE public.neon_worm_leaderboard ENABLE ROW LEVEL SECURITY;
    
    -- 4. RLS 읽기/쓰기 정책 허용
    DROP POLICY IF EXISTS "Allow public read" ON public.neon_worm_leaderboard;
    CREATE POLICY "Allow public read" ON public.neon_worm_leaderboard FOR SELECT USING (true);
    
    DROP POLICY IF EXISTS "Allow public insert" ON public.neon_worm_leaderboard;
    CREATE POLICY "Allow public insert" ON public.neon_worm_leaderboard FOR INSERT WITH CHECK (true);
    
    DROP POLICY IF EXISTS "Allow public update" ON public.neon_worm_leaderboard;
    CREATE POLICY "Allow public update" ON public.neon_worm_leaderboard FOR UPDATE USING (true) WITH CHECK (true);
