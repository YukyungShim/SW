    -- =========================================================================
    -- 1. 네온 웜 랭킹 테이블 생성
    -- =========================================================================
    CREATE TABLE IF NOT EXISTS public.neon_worm_leaderboard (
        id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
        name VARCHAR(12) NOT NULL UNIQUE,
        time_ms INTEGER NOT NULL CHECK (time_ms >= 0),
        apples INTEGER NOT NULL DEFAULT 0 CHECK (apples >= 0),
        score INTEGER NOT NULL DEFAULT 0 CHECK (score >= 0 AND score <= 9999999),
        hazard_level SMALLINT NOT NULL DEFAULT 1 CHECK (hazard_level >= 1),
        played_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
    );
    
    -- =========================================================================
    -- 2. 생존 시간 정렬 인덱스
    -- =========================================================================
    CREATE INDEX IF NOT EXISTS idx_neon_worm_survival_rank 
    ON public.neon_worm_leaderboard (time_ms DESC, apples DESC, played_at ASC);
    
    -- =========================================================================
    -- 3. RLS 활성화 & 읽기/쓰기 보안 정책 (삭제 DELETE는 차단)
    -- =========================================================================
    ALTER TABLE public.neon_worm_leaderboard ENABLE ROW LEVEL SECURITY;
    
    DROP POLICY IF EXISTS "Allow public read access" ON public.neon_worm_leaderboard;
    CREATE POLICY "Allow public read access" 
    ON public.neon_worm_leaderboard
    FOR SELECT 
    USING (true);
    
    DROP POLICY IF EXISTS "Allow public insert and update" ON public.neon_worm_leaderboard;
    DROP POLICY IF EXISTS "Allow public insert" ON public.neon_worm_leaderboard;
    DROP POLICY IF EXISTS "Allow public update" ON public.neon_worm_leaderboard;
    
    CREATE POLICY "Allow public insert" 
    ON public.neon_worm_leaderboard
    FOR INSERT 
    WITH CHECK (true);
    
    CREATE POLICY "Allow public update" 
    ON public.neon_worm_leaderboard
    FOR UPDATE 
    USING (true)
    WITH CHECK (true);
    
    -- =========================================================================
    -- 4. 스마트 랭킹 자동 갱신 및 상태 반환 RPC 함수
    -- =========================================================================
    CREATE OR REPLACE FUNCTION public.save_neon_worm_score(
        p_name VARCHAR(12),
        p_time_ms INTEGER,
        p_apples INTEGER,
        p_score INTEGER,
        p_hazard_level SMALLINT
    )
    RETURNS JSON
    LANGUAGE plpgsql
    SECURITY DEFINER
    AS $$
    DECLARE
        v_existing RECORD;
    BEGIN
        -- 1. 해당 닉네임의 기존 기록 조회
        SELECT * INTO v_existing 
        FROM public.neon_worm_leaderboard 
        WHERE UPPER(name) = UPPER(p_name);
    
        -- 2. 신규 파일럿인 경우: 즉시 INSERT
        IF NOT FOUND THEN
            INSERT INTO public.neon_worm_leaderboard (name, time_ms, apples, score, hazard_level, played_at)
            VALUES (p_name, p_time_ms, p_apples, p_score, p_hazard_level, NOW());
    
            RETURN json_build_object(
                'status', 'inserted',
                'is_new_best', true,
                'time_ms', p_time_ms
            );
        END IF;
    
        -- 3. 기존 기록이 있는 경우: 생존 시간이 더 긴 경우에만 UPDATE
        IF p_time_ms > v_existing.time_ms THEN
            UPDATE public.neon_worm_leaderboard
            SET 
                time_ms = p_time_ms,
                apples = p_apples,
                score = p_score,
                hazard_level = p_hazard_level,
                played_at = NOW()
            WHERE id = v_existing.id;
    
            RETURN json_build_object(
                'status', 'updated',
                'is_new_best', true,
                'time_ms', p_time_ms,
                'prev_time_ms', v_existing.time_ms
            );
        ELSE
            -- 기존 기록이 더 높아 유지된 경우
            RETURN json_build_object(
                'status', 'kept',
                'is_new_best', false,
                'time_ms', v_existing.time_ms
            );
        END IF;
    END;
    $$;
    
    -- 익명(anon) 클라이언트의 RPC 함수 실행 권한 명시적 부여
    GRANT EXECUTE ON FUNCTION public.save_neon_worm_score(VARCHAR, INTEGER, INTEGER, INTEGER, SMALLINT) TO anon,
  authenticated;
